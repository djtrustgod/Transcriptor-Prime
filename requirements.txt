# Core runtime dependencies.
#
# faster-whisper brings in ctranslate2 (the inference engine), tokenizers,
# huggingface-hub (model download), onnxruntime (Silero VAD) and av (PyAV).
#
# PyAV ships FFmpeg's libraries inside its wheel, which is why this app can
# decode MP3/MP4/MKV/etc. without a separate ffmpeg install on the system.
faster-whisper==1.2.1

# CustomTkinter draws the widgets. It brings darkdetect, which reads the Windows
# AppsUseLightTheme registry value -- that is how the app follows the system
# light/dark setting -- plus packaging. Pure Python; no compiled extension.
customtkinter==6.0.0

# Speaker identification ("who said what"): pyannote segmentation plus a speaker
# embedding model, run through ONNX -- no PyTorch. Only ever imported by
# diarize_worker.py, in a child process; see "Why a subprocess" in
# ARCHITECTURE.md.
#
# Both lines are needed and must match. sherpa-onnx is a thin binding; the
# native libraries, including its own onnxruntime.dll, live in sherpa-onnx-core.
# If the core wheel goes missing the binding silently picks up the old
# onnxruntime.dll that ships in Windows' System32 and fails much later with an
# unrelated-looking "API version" error.
sherpa-onnx==1.13.8
sherpa-onnx-core==1.13.8
